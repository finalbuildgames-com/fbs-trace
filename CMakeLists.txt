cmake_minimum_required(VERSION 3.16)
project(FinalBuildTrace VERSION 0.1.0 LANGUAGES C)
include(GNUInstallDirs)
include(CTest)
option(FBS_BUILD_TESTS "Build module tests" ON)
option(FBS_BUILD_EXAMPLES "Build the example" ON)
add_library(fbs_trace src/trace.c)
add_library(fbs::trace ALIAS fbs_trace)
set_target_properties(fbs_trace PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES
  C_EXTENSIONS NO POSITION_INDEPENDENT_CODE ON WINDOWS_EXPORT_ALL_SYMBOLS ON EXPORT_NAME trace)
target_include_directories(fbs_trace PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
if(NOT MSVC)
  target_link_libraries(fbs_trace PUBLIC m)
  target_compile_options(fbs_trace PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
endif()
if(FBS_BUILD_TESTS AND BUILD_TESTING)
  add_executable(fbs_test_trace tests/test_trace.c)
  set_target_properties(fbs_test_trace PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
  target_link_libraries(fbs_test_trace PRIVATE fbs::trace)
  if(NOT MSVC)
    target_compile_options(fbs_test_trace PRIVATE -UNDEBUG -ffp-contract=off)
  endif()
  add_test(NAME trace COMMAND fbs_test_trace)
  set_tests_properties(trace PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/tests)
endif()
if(FBS_BUILD_EXAMPLES)
  add_executable(fbs_trace_example examples/basic.c)
  set_target_properties(fbs_trace_example PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
  target_link_libraries(fbs_trace_example PRIVATE fbs::trace)
  if(BUILD_TESTING)
    add_test(NAME trace_example COMMAND fbs_trace_example)
  endif()
endif()
install(TARGETS fbs_trace EXPORT FinalBuildTraceTargets)
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT FinalBuildTraceTargets NAMESPACE fbs::
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/FinalBuildTrace)
install(FILES LICENSE THIRD_PARTY_NOTICES DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/fbs-trace)
install(DIRECTORY third_party/ DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/fbs-trace/upstream OPTIONAL)
